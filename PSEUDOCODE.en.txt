CLASSROOM NOISE METER

1. NoiseMeter.java
------------------

ALGORITHM ProcessSoundFrame
INPUT  lowest and highest microphone readings, switch position, current time
OUTPUT current sound level, number of LEDs to light, alarm sequence number

BEGIN
    IF low < 0 OR high > 1023 OR low > high OR switchPosition outside -1..3 THEN
        RETURN error 400, "Invalid sensor frame"
    ENDIF

    IF recording AND 60 seconds have elapsed THEN
        CALL FinishSession()
    ENDIF
    IF NOT recording AND the physical switch selects a mode THEN
        Apply the selected mode and reset the warning timer if the mode changed
    ENDIF

    IF currentTime >= ignoreSoundUntil THEN
        span ← high - low
        IF source = DEMO THEN
            span ← CLAMP(span + RANDOM_INTEGER(-3, 3), 0, 1023)
        ENDIF
        reading ← ROUND((span - 3) * 100 / (fullScale - 3))
        level ← MovingAverage(CLAMP(reading, 0, 100))
        CALL CheckWarning(level, currentTime)
    ENDIF

    IF recording THEN
        second ← FLOOR((currentTime - startTime) / 1000)
        readings[second] ← MAX(readings[second], level)
    ENDIF

    lights ← 0
    FOR EACH threshold IN [6, 23, 42, 62, 82]
        IF level >= threshold THEN
            lights ← lights + 1
        ENDIF
    ENDFOR
    RETURN level, lights, alarmSequence
END

FUNCTION MovingAverage(reading)
    IF the window contains 4 readings THEN
        Remove the oldest reading
    ENDIF
    Add reading to the window
    RETURN ROUND(SUM(window) / SIZE(window))
ENDFUNCTION

PROCEDURE CheckWarning(level, currentTime)
    IF level <= limit[currentMode] THEN
        loudSince ← NONE
        warningSent ← FALSE
    ELSE
        IF loudSince = NONE THEN
            loudSince ← currentTime
        ENDIF
        IF currentTime - loudSince >= 5000 AND warningSent = FALSE THEN
            alarmSequence ← alarmSequence + 1
            warningSent ← TRUE
            IF recording THEN
                warnings ← warnings + 1
            ENDIF
            IF source = BOARD THEN
                ignoreSoundUntil ← currentTime + alarmGuardMs
            ENDIF
        ENDIF
    ENDIF
ENDPROCEDURE

The browser plays the selected alarm when alarmSequence changes. Times are in
milliseconds. The initial fullScale is 50. The default limits are 30 for Exam,
50 for Group Work, and 70 for Break.

PROCEDURE SetThreshold(requestedMode, newLimit)
    IF another tab controls the input OR recording THEN
        RETURN error 409
    ENDIF
    IF requestedMode differs from currentMode THEN
        RETURN error 409
    ENDIF
    IF newLimit is not an integer from 0 to 100 THEN
        RETURN error 400
    ENDIF
    IF newLimit differs from limit[currentMode] THEN
        limit[currentMode] ← newLimit
        loudSince ← NONE
        warningSent ← FALSE
    ENDIF
ENDPROCEDURE

PROCEDURE StartSession()
    IF caller does not own the input OR recording OR no recent reading exists THEN
        RETURN error 409
    ENDIF
    FOR i ← 0 TO 59
        readings[i] ← -1
    ENDFOR
    startTime ← currentTime
    warnings ← 0
    loudSince ← NONE
    warningSent ← FALSE
    recording ← TRUE
ENDPROCEDURE

PROCEDURE FinishSession()
    recording ← FALSE
    total ← 0
    FOR i ← 0 TO 59
        IF readings[i] = -1 THEN
            RETURN "Session discarded: missing sensor data"
        ENDIF
        total ← total + readings[i]
    ENDFOR

    average ← total / 60
    top ← FindLoudest(readings)
    peak ← readings[top[0]]
    Shift previous sessions down one row in history[12][60], oldest last
    Copy readings into history[0] and save the matching session summary
    OUTPUT average, peak, warnings
    FOR EACH index IN top
        OUTPUT "Second", index + 1, "Level", readings[index]
    ENDFOR
ENDPROCEDURE

FUNCTION FindLoudest(readings)
    selected ← array of 60 FALSE values
    FOR rank ← 0 TO 2
        best ← NONE
        FOR i ← 0 TO 59
            IF NOT selected[i] THEN
                IF best = NONE OR readings[i] > readings[best] THEN
                    best ← i
                ENDIF
            ENDIF
        ENDFOR
        top[rank] ← best
        selected[best] ← TRUE
    ENDFOR
    RETURN top
ENDFUNCTION

2. NoiseServer.java
-------------------

ALGORITHM RunLocalServer
BEGIN
    port ← supplied port, or 8765 if omitted
    Create one shared noise meter
    Start the HTTP server on 127.0.0.1 at port
    OUTPUT the local URL
    WHILE the server is running
        Accept a request and dispatch it to HandleRequest
    ENDWHILE
END

FUNCTION HandleRequest(request)
    IF host or port does not match the local server THEN
        RETURN HTTP 403
    ENDIF

    IF request is GET /api/state THEN
        action ← "state"
    ELSE IF request is POST /api/command THEN
        IF origin does not match the local page THEN
            RETURN HTTP 403
        ENDIF
        action ← requested command
    ELSE IF request is GET for the home page, stylesheet, or JavaScript file THEN
        IF the file is missing from the JAR THEN
            RETURN HTTP 500
        ENDIF
        RETURN HTTP 200 with the file
    ELSE
        RETURN HTTP 404
    ENDIF

    IF query exceeds 512 characters OR tab ID is invalid THEN
        RETURN HTTP 400
    ENDIF
    Pass action, tab ID, arguments and elapsed time to the meter
    IF an argument is invalid THEN
        RETURN HTTP 400 with the reason
    ELSE IF the action conflicts with the meter's state THEN
        RETURN HTTP 409 with the reason
    ELSE
        RETURN HTTP 200 with the meter state as JSON
    ENDIF
ENDFUNCTION
